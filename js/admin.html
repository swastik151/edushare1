import { db } from "./firebase.js"; import { ref, push } from "https://www.gstatic.com/firebasejs/12.1.0/firebase-database.js"; const postBtn = document.getElementById("postBtn"); const announcement = document.getElementById("announcement"); const status = document.getElementById("status"); postBtn.addEventListener("click", async () => { const text = announcement.value.trim(); if (!text) { status.textContent = "Please type an announcement."; return; } try { await push(ref(db, "announcements"), { text: text, time: Date.now() }); announcement.value = ""; status.textContent = "✅ Announcement posted!"; } catch (err) { console.error(err); status.textContent = "❌ Failed to post."; } });
